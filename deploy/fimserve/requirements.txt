# pip/uv install overrides applied to the standalone fimserve_viewer install
# (see Dockerfile). These force versions regardless of what transitive
# deps request.

numpy==1.26.4

# scipy 1.18+ requires numpy>=2.0 and uses np.long (absent in numpy 1.26) -> fimserve
# import crashes. The standalone image has only fimserve to constrain scipy, so it
# floats to the latest without this cap; 1.17.x is the numpy-1.26-compatible line the
# portal image runs.
scipy<1.18

zarr>=3,<3.1

# Security pins - bump transitive deps off versions flagged High by the Grype
# scan (GHSA/CVE fixes). Capped to the fix's major line so the resolver can't
# overshoot into an incompatible major (e.g. protobuf 7.x breaks google libs).
aiohttp>=3.13.3        # GHSA-6mq8-rvhq-8wgg (High) + others; aiobotocore/s3fs/gcsfs take it unbounded
protobuf>=5.29.6,<6    # GHSA-8qvm-5x2c-j2w7 / GHSA-7gcm-g887-7qv7 (High); google libs require <6

# Hold the google-cloud/grpc stack at its last protobuf-5-gencode versions. Newer
# releases ship protobuf-6 gencode and require protobuf>=6.33.5, which breaks the
# protobuf<6 pin above (teehr needs 5.x) -> fimserve import crashes with
# "gencode 6.33.5 runtime 5.29.6". Pulled transitively via geemap/earthengine-api,
# gcsfs and django-storages.
googleapis-common-protos<1.70
google-api-core<2.25
google-cloud-storage<3
google-cloud-storage-control<1.4
grpc-google-iam-v1<0.14
grpcio-status<1.71
urllib3>=2.6.3,<3      # GHSA-gm62/38jv/2xpw (High); only 2.x carries the fixes
pyarrow>=23.0.1,<24    # GHSA-rgxp-2hwp-jwgg (High); no fix before 23.x
lxml>=6.1.0,<7         # GHSA-vfmq-68hx-4jfw (High); no fix before 6.x
